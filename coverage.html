type ZenityInterface interface {
        SelectFile(extension string) (string, error)
        SelectDirectory() (string, error)
}

type RealZenity struct{}

func (RealZenity) SelectFile(extension string) (string, error) {
        return zenity.SelectFile(
                zenity.FileFilters{
                        // 指定拡張子
                        zenity.FileFilter{
                                Patterns: []string{"*." + extension},
                        },
                        // 全てのファイル
                        zenity.FileFilter{
                                Patterns: []string{"*"},
                        },
                },
                // カレントディレクトリから始める
                zenity.Filename("."),
        )
}

func (RealZenity) SelectDirectory() (string, error) {
        return zenity.SelectFile(
                zenity.Directory(),
                // カレントディレクトリから始める
                zenity.Filename("."),
        )
}

type OrderInfo struct {
        Company        string
        FamilyName     string
        OrderNo        string
        Issue          string
        ProjectCode    string
        Customer       string
        DepartmentCode string
        // Amount         uint
        // Unit           string
        StartDate      time.Time
        EndDate        time.Time
        FullName       string
        Representative string
}

func ToOrderInfo(record map[string]interface{}) (OrderInfo, error) {
        var ret OrderInfo

        // 文字列型
        for colName, retv := range map[string]*string{
                CompanyColName:        &ret.Company,
                FamilyNameColName:     &ret.FamilyName,
                OrderNoColName:        &ret.OrderNo,
                IssueColName:          &ret.Issue,
                ProjectCodeColName:    &ret.ProjectCode,
                CustomerColName:       &ret.Customer,
                DepartmentCodeColName: &ret.DepartmentCode,
                FullNameColName:       &ret.FullName,
                RepresentativeColName: &ret.Representative,
        } {
                if v, ok := record[colName]; ok {
                        if strVal, ok := v.(string); ok {
                                *retv = strVal
                        } else {
                                return OrderInfo{}, fmt.Errorf("%v is not string", strVal)
                        }
                } else {
                        return OrderInfo{}, fmt.Errorf("there is no key: %v", colName)
                }
        }

        // 日時型
        for colName, retv := range map[string]*time.Time{
                StartDateColName: &ret.StartDate,
                EndDateColName:   &ret.EndDate,
        } {
                if v, ok := record[colName]; ok {
                        if strVal, ok := v.(string); ok {
                                dt, err := time.Parse("20060102", strVal)
                                if err != nil {
                                        return OrderInfo{}, err
                                }
                                *retv = dt
                        } else {
                                return OrderInfo{}, fmt.Errorf("%v is not string", v)
                        }
                } else {
                        return OrderInfo{}, fmt.Errorf("there is no key: %v", colName)
                }
        }

        return ret, nil
}

func FirstSheet(
        outFile *excelize.File,
        orderInfo OrderInfo,
) error {
        // 1セル1文字ずつ入力する情報
        for _, v := range []struct {
                leftmostCell string
                str          string
        }{
                {leftmostCell: "D18", str: orderInfo.ProjectCode},
                {leftmostCell: "D19", str: orderInfo.DepartmentCode},
        } {
                // 左端のセル名を取得
                col, row, err := excelize.CellNameToCoordinates(v.leftmostCell)
                if err != nil {
                        fmt.Println(err)
                        continue
                }
                // 左から1文字ずつ記入
                for colShift, ch := range v.str {
                        cellName, err := excelize.CoordinatesToCellName(
                                col+colShift, row)
                        if err != nil {
                                fmt.Println(err)
                                continue
                        }
                        if err := outFile.SetCellValue(
                                BeforeUseSheetName, cellName, string(ch)); err != nil {
                                fmt.Println(err)
                        }
                }
        }

        // 1セルに文字列として入力する情報
        for _, v := range []struct {
                cell string
                str  string
        }{
                {cell: "D20", str: orderInfo.Customer},
                {cell: "D21", str: orderInfo.Issue},
        } {
                if err := outFile.SetCellValue(
                        BeforeUseSheetName, v.cell, v.str,
                ); err != nil {
                        fmt.Println(err)
                }
        }

        return nil
}

func SecondSheet(
        outFile *excelize.File,
        orderInfo OrderInfo,
        cacheDirIF CacheDirInterface,
) error {
        workDays, err := retrieveWorkDays(orderInfo.StartDate, orderInfo.EndDate, cacheDirIF)
        if err != nil {
                fmt.Println(err)
        }

        // 勤務表の上部・下部
        for _, v := range []struct {
                cell  string
                value interface{}
        }{
                {cell: "G9", value: orderInfo.StartDate.Year()},
                {cell: "K9", value: int(orderInfo.StartDate.Month())},
                {cell: "E14", value: "S-" + orderInfo.OrderNo},
                {cell: "S16", value: orderInfo.FullName},
                {cell: "S60", value: orderInfo.Representative},
        } {
                if err := outFile.SetCellValue(OutsourceSheetName, v.cell, v.value); err != nil {
                        fmt.Println(err)
                }
        }

        // 各勤務日
        for row := 20; row <= 50; row++ {
                dayStr, err := outFile.GetCellValue(
                        OutsourceSheetName, "B"+strconv.Itoa(row))
                if err != nil {
                        fmt.Println(err)
                        continue
                }
                day, err := strconv.Atoi(dayStr)
                if err != nil {
                        fmt.Println(err)
                        continue
                }

                // 1行分埋める
                for _, v := range []struct {
                        colName string
                        value   interface{}
                }{
                        {colName: "C", value: time.Duration(8) * time.Hour},
                        {colName: "N", value: orderInfo.Issue},
                } {
                        // 勤務日でないならばセルを空とする
                        if !slices.Contains(workDays, day) {
                                v.value = nil
                        }

                        if err := outFile.SetCellValue(
                                OutsourceSheetName, v.colName+strconv.Itoa(row), v.value,
                        ); err != nil {
                                fmt.Println(err)
                        }
                }
        }

        return nil
}

func createEachSpreadSheet(
        record map[string]interface{},
        pathOf PathOf,
        cacheDirIF CacheDirInterface,
) error {
        orderInfo, err := ToOrderInfo(record)
        if err != nil {
                return err
        }

        outPath := pathOf.OutputDir + fmt.Sprintf(
                "/S-%s【%s】%s_%04d%02d.xlsx",
                orderInfo.OrderNo,
                orderInfo.Company,
                orderInfo.FamilyName,
                orderInfo.StartDate.Year(), orderInfo.StartDate.Month())
        if err := copyFile(pathOf.TemplateXlsx, outPath); err != nil {
                return err
        }
        outFile, err := excelize.OpenFile(outPath)
        if err != nil {
                return err
        }
        defer outFile.Close()

        // 「ご使用の前に」シート
        if err := FirstSheet(outFile, orderInfo); err != nil {
                return err
        }

        // 「システム要員作業実績報告(委任)」シート
        if err := SecondSheet(outFile, orderInfo, cacheDirIF); err != nil {
                return err
        }

        // セル再計算の後、編集結果を保存
        if err := outFile.UpdateLinkedValue(); err != nil {
                return err
        }
        if err := outFile.Save(); err != nil {
                return err
        }

        fmt.Printf("%s is saved.\n", outPath)
        return nil
}

func copyFile(src, dst string) error {
        sourceFile, err := os.Open(src)
        if err != nil {
                return err
        }
        defer sourceFile.Close()

        destFile, err := os.Create(dst)
        if err != nil {
                return err
        }
        defer destFile.Close()

        _, err = io.Copy(destFile, sourceFile)
        return err
}

func fileExists(path string) bool {
        _, err := os.Stat(path)
        return !errors.Is(err, os.ErrNotExist)
}

func makeCacheDir(cacheDirIF CacheDirInterface) error {
        cacheDir, err := CacheDir(cacheDirIF)
        if err != nil {
                return err
        }
        if !fileExists(cacheDir) {
                if err := os.MkdirAll(cacheDir, 0755); err != nil {
                        return err
                }
        }

        return nil
}

func getOrderTable(path string) (dataframe.DataFrame, error) {
        // スプレッドシートを開く
        f, err := excelize.OpenFile(path)
        if err != nil {
                return dataframe.DataFrame{}, err
        }
        defer f.Close()

        // 最初のシートのデータの範囲を取得する
        sheetName := f.GetSheetName(0)
        dim, err := f.GetSheetDimension(sheetName)
        if err != nil {
                return dataframe.DataFrame{}, err
        }
        topLeftCol, topLeftRow, err := excelize.CellNameToCoordinates(strings.Split(dim, ":")[0])
        if err != nil {
                return dataframe.DataFrame{}, err
        }

        // 最初のシートのテーブルを行ごとに読み込む
        var records [][]string
        rows, err := f.GetRows(sheetName)
        if err != nil {
                return dataframe.DataFrame{}, err
        }
        for _, row := range rows[topLeftRow-1:] {
                // 空の行は読み飛ばす
                // (GetSheetDimension()の不具合か、空行にもかかわらず範囲に含まれる場合があるため)
                if len(strings.Join(row, "")) == 0 {
                        continue
                }

                var record []string
                for _, cell := range row[topLeftCol-1:] {
                        // セルの値の両端の空白, CR, LFを除去した値を格納する
                        record = append(record, strings.TrimSpace(strings.NewReplacer("\r", "", "\n", "").Replace(cell)))
                }

                records = append(records, record)
        }

        return dataframe.LoadRecords(
                records, dataframe.DetectTypes(false),
                dataframe.NaNValues([]string{})), nil
}

func confirm(reader io.Reader, prompt string) bool {
        bufReader := bufio.NewReader(reader)
        for {
                fmt.Printf("%s [y/n]: ", prompt)
                response, err := bufReader.ReadString('\n')
                if err != nil {
                        log.Fatal(err)
                }

                response = strings.ToLower(strings.TrimSpace(response))

                if response == "y" || response == "yes" {
                        return true
                } else if response == "n" || response == "no" {
                        return false
                }
        }
}

func readPathsFromJSON(cacheDirIF CacheDirInterface) (PathOf, error) {
        pathsJSONPath, err := PathsJSONPath(cacheDirIF)
        if err != nil {
                return PathOf{}, err
        }
        // 前回実行時のパス情報を開く
        pathsCacheFile, err := os.Open(pathsJSONPath)
        if err != nil {
                return PathOf{}, err
        }
        defer pathsCacheFile.Close()

        // 前回実行時のパス情報を読み込む
        pathsCacheBytes, err := io.ReadAll(pathsCacheFile)
        if err != nil {
                return PathOf{}, err
        }

        var pathOf PathOf
        if err := json.NewDecoder(strings.NewReader(string(pathsCacheBytes))).Decode((&pathOf)); err != nil {
                return PathOf{}, err
        }

        // 入力情報となるパスについては、ファイル自身が存在しないものは捨てる
        for _, path := range []*string{
                &pathOf.OrderXlsx,
                &pathOf.NamesCSV,
                &pathOf.TemplateXlsx,
        } {
                if !fileExists(*path) {
                        *path = ""
                }
        }

        return pathOf, nil
}

func inputPaths(appIF *AppInterfaces) (PathOf, error) {
        // 前回実行時のパス情報を読み込む
        pathOf, _ := readPathsFromJSON(appIF.CacheDirIF)

        // 発注情報の入力を促す
        bufReader := bufio.NewReader(appIF.InputReader)
        for _, v := range []struct {
                fileDescription string
                extension       string
                path            *string
        }{
                {
                        fileDescription: "発注情報が記載されたxlsxファイル",
                        extension:       "xlsx",
                        path:            &pathOf.OrderXlsx,
                },
                {
                        fileDescription: "名簿となるcsvファイル(姓,姓名のテーブル)",
                        extension:       "csv",
                        path:            &pathOf.NamesCSV,
                },
                {
                        fileDescription: "勤務表テンプレートとなるxlsxファイル",
                        extension:       "xlsx",
                        path:            &pathOf.TemplateXlsx,
                },
                {
                        fileDescription: "各作業の勤務表出力ディレクトリ",
                        path:            &pathOf.OutputDir,
                },
        } {
                doUseCache := false
                if *v.path != "" {
                        doUseCache = confirm(
                                appIF.InputReader,
                                fmt.Sprintf(
                                        "%sは\n %s\nでよろしいですか?",
                                        v.fileDescription, *v.path,
                                ),
                        )
                }
                if !doUseCache {
                        fmt.Printf("%sを選択してください。(Enter押下でファイル選択に進みます)", v.fileDescription)
                        _, err := bufReader.ReadString('\n')
                        if err != nil {
                                fmt.Println(err)
                                continue
                        }
                        if v.extension != "" {
                                *v.path, err = appIF.ZenityIF.SelectFile(v.extension)
                        } else {
                                *v.path, err = appIF.ZenityIF.SelectDirectory()
                        }
                        if err != nil {
                                fmt.Println(err)
                                continue
                        }
                        fmt.Println(*v.path)
                }
        }

        // キャッシュにパス情報を記録する
        jsonPath, err := PathsJSONPath(appIF.CacheDirIF)
        if err != nil {
                fmt.Println(err)
        } else {
                jsonBytes, err := json.Marshal(pathOf)
                if err != nil {
                        fmt.Println(err)
                } else if err := os.WriteFile(jsonPath, jsonBytes, 0644); err != nil {
                        fmt.Println(err)
                }
        }

        return pathOf, nil
}

func joinFullNames(namesPath string, orderDf dataframe.DataFrame) (dataframe.DataFrame, error) {
        // 名簿を読み込み左結合
        csvFile, err := os.Open(namesPath)
        if err != nil {
                return orderDf, err
        }
        defer csvFile.Close()

        csvStr, err := io.ReadAll(csvFile)
        if err != nil {
                return orderDf, err
        }

        if !slices.Contains(orderDf.Names(), FamilyNameColName) {
                return orderDf, fmt.Errorf(`発注情報に"%s"の列がありません`, FamilyNameColName)
        }
        return orderDf.LeftJoin(
                dataframe.ReadCSV(
                        strings.NewReader(string(csvStr))), FamilyNameColName), nil
}

func run(appIF *AppInterfaces) error {
        // キャッシュディレクトリを作成する
        if err := makeCacheDir(appIF.CacheDirIF); err != nil {
                fmt.Println(err)
        }

        // 発注情報、名簿、テンプレート、出力フォルダをユーザに入力してもらう
        pathOf, err := inputPaths(appIF)
        if err != nil {
                return err
        }

        // 発注情報を取り込む
        orderDf, err := getOrderTable(pathOf.OrderXlsx)
        if err != nil {
                return err
        }

        // 名簿を発注情報に合併
        joinedDf, err := joinFullNames(pathOf.NamesCSV, orderDf)
        if err != nil {
                return err
        }

        // 出力先ディレクトリが存在しない場合は作成する
        if err := os.MkdirAll(pathOf.OutputDir, 0666); err != nil {
                return err
        }

        // 発注情報についてループ
        errors := make(chan error, joinedDf.Nrow())
        var waitGroup sync.WaitGroup
        for _, record := range joinedDf.Maps() {
                waitGroup.Add(1)
                go func(r map[string]interface{}) {
                        defer waitGroup.Done()
                        errors <- createEachSpreadSheet(r, pathOf, appIF.CacheDirIF)
                }(record)
        }
        go func() {
                waitGroup.Wait()
                close(errors)
        }()

        for err := range errors {
                if err != nil {
                        fmt.Println(err)
                }
        }

        fmt.Println("完了しました。Enterで終了します。")
        _, err = bufio.NewReader(appIF.InputReader).ReadString('\n')
        if err != nil {
                fmt.Println(err)
        }

        return nil
}

// APIのレスポンス内の日付用の型
// RFC2339形式の日時文字列ではないため。
type CustomDate struct {
        time.Time
}

// JSONデコード時の日付パース
func (cd *CustomDate) UnmarshalJSON(b []byte) error {
        date, err := time.Parse("2006-01-02", strings.Trim(string(b), "\""))
        if err != nil {
                return err
        }
        cd.Time = date
        return nil
}

type HolidaysJSON struct {
        Holidays []struct {
                Date CustomDate
                Name string
        }
}

func readHolidaysFromJSON(date time.Time, cacheDirIF CacheDirInterface) (HolidaysJSON, error) {
        jsonPath, err := HolidaysJSONPath(cacheDirIF)
        if err != nil {
                return HolidaysJSON{}, err
        }
        // キャッシュを開く
        jsonFile, err := os.Open(jsonPath)
        if err != nil {
                return HolidaysJSON{}, err
        }
        defer jsonFile.Close()

        // キャッシュを読む
        jsonBytes, err := io.ReadAll(jsonFile)
        if err != nil {
                return HolidaysJSON{}, err
        }

        var holidaysJSON HolidaysJSON
        if err := json.NewDecoder(
                strings.NewReader(string(jsonBytes)),
        ).Decode(&holidaysJSON); err != nil {
                return HolidaysJSON{}, err
        }

        // キャッシュが当月分のものであるか確認する
        if len(holidaysJSON.Holidays) == 0 {
                return HolidaysJSON{}, fmt.Errorf("JSONが不正")
        }
        dateFromJSON := holidaysJSON.Holidays[0].Date

        if dateFromJSON.Year() != date.Year() || dateFromJSON.Month() != date.Month() {
                return HolidaysJSON{}, fmt.Errorf("当月分の祝日データではない")
        }

        return holidaysJSON, nil
}

func retrieveWorkDays(
        startDate time.Time,
        endDate time.Time,
        cacheDirIF CacheDirInterface,
) ([]int, error) {
        // キャッシュを調べる
        holidaysJSON, err := readHolidaysFromJSON(startDate, cacheDirIF)
        if err != nil {
                // キャッシュからの取得に失敗した場合は
                // https://github.com/shogo82148/holidays-jp
                // によるAPIにより当月の祝日を取得する
                response, err := http.Get(
                        fmt.Sprintf(
                                HolidaysURL+"/%04d/%02d",
                                startDate.Year(), startDate.Month()))
                if err != nil {
                        return []int{}, err
                }
                defer response.Body.Close()

                // 取得結果をパースする
                body, err := io.ReadAll(response.Body)
                if err != nil {
                        return []int{}, err
                }

                if err := json.NewDecoder(strings.NewReader(string(body))).Decode(&holidaysJSON); err != nil {
                        return []int{}, err
                }

                // リクエスト結果をキャッシュする
                jsonPath, err := HolidaysJSONPath(cacheDirIF)
                if err != nil {
                        fmt.Println(err)
                } else {
                        if err := os.WriteFile(jsonPath, body, 0644); err != nil {
                                fmt.Println(err)
                        }
                }
        }

        // 勤務日のリストを作成する
        var workDays []int
daysLoop:
        for day := startDate.Day(); day <= endDate.Day(); day++ {
                // 存在しない日付ならばスキップ
                date, err := time.Parse(
                        "20060102",
                        fmt.Sprintf(
                                "%04d%02d%02d",
                                startDate.Year(), startDate.Month(), day),
                )
                if err != nil {
                        continue
                }

                // 土日ならばスキップ
                if slices.Contains([]time.Weekday{time.Saturday, time.Sunday}, date.Weekday()) {
                        continue
                }

                // 祝日ならばスキップ
                for _, holiday := range holidaysJSON.Holidays {
                        if holiday.Date.Day() == date.Day() {
                                continue daysLoop
                        }
                }

                // 勤務日として追加
                workDays = append(workDays, day)
        }

        return workDays, nil
}
