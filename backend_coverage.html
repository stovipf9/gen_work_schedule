import (

        "os"

        "github.com/spf13/afero"
)

type SLogger struct {
        slog.Logger
        logFile afero.File
}

func NewSLogger(fs afero.Fs, logFileName string) *SLogger {
        stdoutLogger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
        logFile, err := fs.OpenFile(
                logFileName, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
        if err != nil {
                stdoutLogger.Warn(err.Error())
                return &SLogger{*stdoutLogger, nil}
        }

        return &SLogger{*slog.New(slog.NewJSONHandler(logFile, nil)), logFile}
}

// logger.Logger interface methods
func (s *SLogger) Print(message string) {
        s.Logger.Debug(message)
}

func (s *SLogger) Trace(message string) {
        s.Logger.Debug(message)
}

func (s *SLogger) Debug(message string) {
        s.Logger.Debug(message)
}

func (s *SLogger) Info(message string) {
        s.Logger.Info(message)
}

func (s *SLogger) Warning(message string) {
        s.Logger.Warn(message)
}

func (s *SLogger) Error(message string) {
        s.Logger.Error(message)
}

func (s *SLogger) Fatal(message string) {
        s.Logger.Error(message)
}

func (s *SLogger) Close() {
        if s.logFile != nil {
                s.logFile.Close()
        }
}

type LogLevel int

const (
        TRACE LogLevel = iota + 1
        DEBUG
        INFO
        WARNING
        ERROR
        FATAL
)

func (s *SLogger) Log(level LogLevel, err error) error {
        if err == nil {
                return nil
        }
        map[LogLevel]func(string){
                TRACE:   s.Trace,
                DEBUG:   s.Debug,
                INFO:    s.Info,
                WARNING: s.Warning,
                ERROR:   s.Error,
                FATAL:   s.Fatal,
        }[level](err.Error())
        return err
}

const (

        HOLIDAYS_JSON = "holidays_%s.json"

type CacheService struct {
        cacheDirFs afero.Fs
}

func NewCacheService(
        cacheDirfs afero.Fs,
        dedicatedCacheDirName string,
) *CacheService {
        return &CacheService{cacheDirFs: afero.NewBasePathFs(cacheDirfs, dedicatedCacheDirName)}
}

func (c CacheService) CacheDirFs() afero.Fs {
        return c.cacheDirFs
}

func (c CacheService) LoadCachedPaths() (map[PathKey]string, error) {
        paths, err := LoadCachedData[map[PathKey]string](
                c.cacheDirFs,
                PATHS_JSON,
        )
        if paths == nil || err != nil {
                return make(map[PathKey]string), err
        }

        return paths, nil
}

func (c CacheService) CachePaths(paths map[PathKey]string) error {
        return CacheData(
                c.cacheDirFs,
                paths,
                PATHS_JSON,
        )
}

func (CacheService) dateToFileName(date time.Time) string {
        return fmt.Sprintf(HOLIDAYS_JSON, date.Format("200601"))
}

func (c CacheService) LoadCachedHolidays(date time.Time) (
        holidays HolidaysJson,
        _ error,
) {
        holidays, err := LoadCachedData[HolidaysJson](
                c.cacheDirFs,
                CacheService{}.dateToFileName(date),
        )
        if err != nil {
                return holidays, err
        }

        // キャッシュが当月分のものであるか確認する
        if len(holidays.Holidays) == 0 {
                // 祝日0日
                return HolidaysJson{}, fmt.Errorf("JSONが不正")
        }
        dateFromJson := holidays.Holidays[0].Date
        if dateFromJson.Year() != date.Year() ||
                dateFromJson.Month() != date.Month() {
                return HolidaysJson{}, fmt.Errorf("当月分の祝日データではない")
        }

        return holidays, nil
}

func (c CacheService) CacheHolidays(rawHolidays []byte) error {

        var holidays HolidaysJson
        if err := json.Unmarshal(rawHolidays, &holidays); err != nil {
                return err
        }
        if len(holidays.Holidays) == 0 {
                return fmt.Errorf("No holidays in %v", rawHolidays)
        }

        return CacheData(
                c.cacheDirFs,
                rawHolidays,
                CacheService{}.dateToFileName(holidays.Holidays[0].Date.Time),
        )
}

func CacheData[T any](
        fs afero.Fs,
        data T,
        jsonPath string,
) error {
        // キャッシュファイルを読み書きロック
        lock := filelm.GetLock(jsonPath)
        lock.Lock()
        defer lock.Unlock()

        // デシリアライズ
        // 既にデータがバイト列なら不要
        jsonBytes, ok := any(data).([]byte)
        if !ok {
                var err error
                jsonBytes, err = json.Marshal(data)
                if err != nil {
                        return err
                }
        }

        if err := fs.MkdirAll(filepath.Dir(jsonPath), 0755); err != nil {
                return err
        }

        // 書き込み
        if err := afero.WriteFile(fs, jsonPath, jsonBytes, 0666); err != nil {
                return err
        }

        return nil
}

func LoadCachedData[T any](
        fs afero.Fs,
        jsonPath string,
) (cachedData T, _ error) {
        lock := filelm.GetLock(jsonPath)
        lock.RLock()
        defer lock.RUnlock()

        // キャッシュを読み込む
        jsonBytes, err := afero.ReadFile(fs, jsonPath)
        if err != nil {
                return cachedData, err
        }

        // シリアライズ
        if err := json.NewDecoder(
                strings.NewReader(string(jsonBytes)),
        ).Decode(&cachedData); err != nil {
                return cachedData, err
        }

        return cachedData, nil
}

func NewEachExcelScheduleMaker(

        return &EachExcelScheduleMaker{

                workdays:         workdays,
                fullName:         fullName,
                outputFilePath:   outputFilePath,
                templateFilePath: templatePath,

func (e *EachExcelScheduleMaker) MakeSchedule() error {
        // テンプレートからコピー
        if err := e.copyFile(e.templateFilePath, e.outputFilePath); err != nil {
                return err
        }

        // 出力ファイルを開く
        outFile, err := excelize.OpenFile(e.outputFilePath)
        if err != nil {
                return err
        }
        defer outFile.Close()

        // 「ご使用の前に」シート
        if err := e.fillBeforeUseSheet(outFile); err != nil {
                return err
        }

        // 「システム要員作業実績報告(委任)」シート
        if err := e.fillOutSourceSheet(outFile); err != nil {
                return err
        }

        // 「御見積書 (委任)」シート
        if err := e.fillQuotDelegationSheet(outFile); err != nil {
                return err
        }

        // セル再計算を行い
        if err := outFile.UpdateLinkedValue(); err != nil {
                return err
        }
        // 編集結果を保存
        if err := outFile.Save(); err != nil {
                return err
        }

        return nil
}

func (e *EachExcelScheduleMaker) fillBeforeUseSheet(
        outFile *excelize.File,
) error {
        // 1セル1文字ずつ入力する情報
        for _, v := range []struct {
                leftmostCell string
                str          string
        }{
                {leftmostCell: "D18", str: e.order.ProjectCode},
                {leftmostCell: "D19", str: e.order.DepartmentCode},
        } {
                // 左端のセル名を取得
                col, row, err := excelize.CellNameToCoordinates(v.leftmostCell)
                if err != nil {
                        return err
                }
                // 左から1文字ずつ記入
                for colShift, ch := range v.str {
                        cellName, err := excelize.CoordinatesToCellName(
                                col+colShift, row,
                        )
                        if err != nil {
                                return err
                        }
                        if err := outFile.SetCellValue(
                                BeforeUseSheetName, cellName, string(ch),
                        ); err != nil {
                                return err
                        }
                }
        }

        // 1セルに文字列として入力する情報
        for _, v := range []struct {
                cell string
                str  string
        }{
                {cell: "D20", str: e.order.Customer},
                {cell: "D21", str: e.order.Issue},
        } {
                if err := outFile.SetCellValue(
                        BeforeUseSheetName, v.cell, v.str,
                ); err != nil {
                        return err
                }
        }

        return nil
}

func (e *EachExcelScheduleMaker) fillOutSourceSheet(
        outFile *excelize.File,
) error {

        // 勤務表の上部・下部
        for _, v := range []struct {
                cell  string
                value any
        }{
                {cell: "G9", value: e.order.StartDate.Year()},
                {cell: "K9", value: int(e.order.StartDate.Month())},
                {cell: "E14", value: "S-" + e.order.OrderNo},
                {cell: "S16", value: e.fullName},
                {cell: "B60", value: e.order.Department},
                {cell: "S60", value: e.order.Representative},
        } {
                if err := outFile.SetCellValue(OutsourceSheetName, v.cell, v.value); err != nil {
                        return err
                }
        }

        // 各勤務日
        for row := 20; row <= 50; row++ {
                dayStr, err := outFile.GetCellValue(
                        OutsourceSheetName, "B"+strconv.Itoa(row))
                if err != nil {
                        return err
                }
                day, err := strconv.Atoi(strings.TrimSpace(dayStr))
                if err != nil {
                        return err
                }

                // 1行分埋める
                for _, v := range []struct {
                        colName string
                        value   any
                }{
                        {colName: "C", value: time.Duration(8) * time.Hour}, //定時想定
                        {colName: "N", value: e.order.Issue},
                } {
                        // 勤務日でないならばセルを空とする
                        if !slices.Contains(e.workdays, day) {
                                v.value = nil
                        }

                        if err := outFile.SetCellValue(
                                OutsourceSheetName, v.colName+strconv.Itoa(row), v.value,
                        ); err != nil {
                                return err
                        }
                }
        }

        return nil
}

func (e *EachExcelScheduleMaker) fillQuotDelegationSheet(
        outFile *excelize.File,
) error {
        return outFile.SetCellValue(
                QuotationDelegationSheetName, "I26", e.order.Department,
        )
}

func (e *EachExcelScheduleMaker) copyFile(src string, dst string) error {
        sourceFile, err := e.fs.Open(src)
        if err != nil {
                return err
        }
        defer sourceFile.Close()

        // コピー先フォルダが存在しない場合は作成
        if err := e.fs.MkdirAll(filepath.Dir(dst), 0666); err != nil {
                return err
        }

        destFile, err := e.fs.Create(dst)
        if err != nil {
                return err
        }
        defer destFile.Close()

        _, err = io.Copy(destFile, sourceFile)
        return err
}

        workdaysRetriever    WorkdaysRetriever
        newEachScheduleMaker EachScheduleMakerFactory
}

        workdaysRetriever WorkdaysRetriever,
        newEachScheduleMaker EachScheduleMakerFactory,
) *GenerateService {
        return &GenerateService{
                workdaysRetriever:    workdaysRetriever,
                newEachScheduleMaker: newEachScheduleMaker,
        }
}

func (g *GenerateService) Generate(
        orders []Order,
        rosterRecords map[string]string,
        outputDirPath string,
        templateFilePath string,
) (eachResults map[string]error, criticalError error) {
        // 発注・名簿を姓で結合してマップとする
        joinedMap := make(map[Order]string, len(orders))
        for _, order := range orders {
                if fullName, exists := rosterRecords[order.FamilyName]; exists {
                        joinedMap[order] = fullName
                } else {
                        return eachResults, fmt.Errorf("名簿に%sさんの姓名がありません", order.FamilyName)
                }
        }

        // 発注情報についてループ
        type EachResult struct {
                fileName string
                error_   error
        }
        resultsChan := make(chan EachResult, len(joinedMap))
        var waitGroup sync.WaitGroup
        for order, fullName := range joinedMap {
                waitGroup.Add(1)
                go func(o Order, f string) {
                        defer waitGroup.Done()
                        fileName := o.GetOutFileName()

                        workdays, err := g.workdaysRetriever.RetrieveWorkdays(
                                o.StartDate.Time, o.EndDate.Time)
                        if err != nil {
                                resultsChan <- EachResult{
                                        fileName: fileName,
                                        error_:   err,
                                }
                                return
                        }

                        resultsChan <- EachResult{
                                fileName: fileName,
                                error_: g.newEachScheduleMaker(&o, workdays, f,
                                        filepath.Join(outputDirPath, fileName),
                                        templateFilePath,
                                ).MakeSchedule(),
                        }

                }(order, fullName)
        }
        go func() {
                waitGroup.Wait()
                close(resultsChan)
        }()

        eachResults = make(map[string]error, len(joinedMap))
        for result := range resultsChan {
                eachResults[result.fileName] = result.error_
        }

        return eachResults, nil
}

import "sync"

type LockManager[T any] struct {
        locks  map[string]*sync.RWMutex // 読み書きロック
        mu     sync.Mutex               // 読み書きロック自体のロック
        genKey func(T) string
}

func NewLockManager[T any](genkey func(T) string) *LockManager[T] {
        return &LockManager[T]{
                locks:  make(map[string]*sync.RWMutex),
                genKey: genkey,
        }
}

func (lm *LockManager[T]) GetLock(key T) *sync.RWMutex {
        lm.mu.Lock()
        defer lm.mu.Unlock()

        keyStr := lm.genKey(key)

        if lock, exists := lm.locks[keyStr]; exists {
                // 既存の読み書きロック
                return lock
        }
        // 新規の読み書きロック
        lock := &sync.RWMutex{}
        lm.locks[keyStr] = lock
        return lock
}

type OrderService struct{}

}

func (OrderService) LoadOrders(path string) (
        orders []Order, _ error,
) {
        // スプレッドシートを開く
        f, err := excelize.OpenFile(path)
        if err != nil {
                return orders, err
        }
        defer f.Close()

        // 最初のシートのデータの範囲を取得する
        sheetName := f.GetSheetName(0)
        dim, err := f.GetSheetDimension(sheetName)
        if err != nil {
                return orders, err
        }
        topLeftCol, topLeftRow, err := excelize.CellNameToCoordinates(strings.Split(dim, ":")[0])
        if err != nil {
                return orders, err
        }

        // 最初のシートのテーブルを行ごとに読み込む
        // デシリアライズの簡単さのため、メモリ上CSV化してから処理する
        rows, err := f.GetRows(sheetName)
        if err != nil {
                return orders, err
        }
        if len(rows) == 0 {
                return orders, fmt.Errorf("発注情報のテーブルサイズが0")
        }
        var buf bytes.Buffer
        w := csv.NewWriter(&buf)
        isWritten := false
        for _, row := range rows[topLeftRow-1:] {
                // 空の行は読み飛ばす
                // (GetSheetDimension()の不具合か、空行にもかかわらず範囲に含まれる場合があるため)
                if len(strings.Join(row, "")) == 0 {
                        if !isWritten {
                                // 前方の空行は読み飛ばすのみ
                                continue
                        } else {
                                // 後方の空行は読み取りを終える
                                break
                        }
                }

                var trimmedRow []string
                for _, cell := range row[topLeftCol-1:] {
                        // セルの値の両端の空白, CR, LFを除去した値を格納する
                        trimmedRow = append(trimmedRow,
                                strings.TrimSpace(strings.NewReplacer("\r", "", "\n", "").Replace(cell)))
                }
                if err := w.Write(trimmedRow); err != nil {
                        return orders, err
                }
                isWritten = true
        }
        w.Flush()

        // CSVデータをデシリアライズ
        if err := csvutil.Unmarshal(buf.Bytes(), &orders); err != nil {
                return orders, err
        }

        return orders, nil
}

func NewRetrieveWorkdaysService(

        return &RetrieveWorkdaysService{

func (r *RetrieveWorkdaysService) RetrieveWorkdays(
        startDate time.Time,
        endDate time.Time,
) (workdays []int, error_ error) {

        // キャッシュを調べる
        holidaysJSON, err := r.loadCachedHolidays(startDate)
        if err != nil {
                r.warnLogger.Warning(err.Error())
                // キャッシュからの取得に失敗した場合はリクエスト
                holidaysJSON_, err := r.requestAndCacheHolidays(startDate)
                if err != nil {
                        return []int{}, err
                }
                holidaysJSON = holidaysJSON_
        }

        // 勤務日のリストを作成する
        // TODO: 先に祝日抜きで作っておいて、祝日取れなかったかも、みたいな状況にするのが自然では？
daysLoop:
        for day := startDate.Day(); day <= endDate.Day(); day++ {
                // 存在しない日付ならばスキップ
                date, err := time.Parse(
                        "20060102",
                        fmt.Sprintf(
                                "%04d%02d%02d",
                                startDate.Year(), startDate.Month(), day),
                )
                if err != nil {
                        // 実質的にはエラー発生しない
                        r.warnLogger.Warning(err.Error())
                        continue
                }

                // 土日ならばスキップ
                if slices.Contains([]time.Weekday{time.Saturday, time.Sunday}, date.Weekday()) {
                        continue
                }

                // 祝日ならばスキップ
                for _, holiday := range holidaysJSON.Holidays {
                        if holiday.Date.Day() == date.Day() {
                                continue daysLoop
                        }
                }

                // 勤務日として追加
                workdays = append(workdays, day)
        }

        return workdays, nil
}

func (r *RetrieveWorkdaysService) requestAndCacheHolidays(
        startDate time.Time,
) (holidays HolidaysJson, _ error) {
        lock := reqlm.GetLock(startDate)
        lock.Lock()
        defer lock.Unlock()

        // https://github.com/shogo82148/holidays-jp
        // によるAPIにより当月の祝日を取得する
        response, err := r.getter.Get(fmt.Sprintf(
                HolidaysURL+"/%04d/%02d",
                startDate.Year(), startDate.Month()))
        if err != nil {
                return holidays, err
        }
        defer response.Body.Close()

        // 取得結果をパース
        body, err := io.ReadAll(response.Body)
        if err != nil {
                return holidays, err
        }
        // リクエスト結果をキャッシュ
        if err := r.holidaysCacher.CacheHolidays(body); err != nil {
                r.warnLogger.Warning(err.Error())
        }

        if err := json.NewDecoder(
                strings.NewReader(string(body))).Decode(&holidays); err != nil {
                return holidays, err
        }

        return holidays, nil
}

func (r *RetrieveWorkdaysService) loadCachedHolidays(date time.Time) (HolidaysJson, error) {
        lock := reqlm.GetLock(date)
        lock.RLock()
        defer lock.RUnlock()

        return r.holidaysCacher.LoadCachedHolidays(date)
}

package cmd

import (
        "fmt"

        "github.com/jszwec/csvutil"
        "github.com/samber/lo"
        "github.com/spf13/afero"
)

type RosterRecord struct {
        FamilyName string `json:"familyName" csv:"姓"`
        FullName   string `json:"fullName" csv:"姓名"`
}

type RosterService struct {
        fs afero.Fs
}

func NewRosterService(fs afero.Fs) *RosterService {
        return &RosterService{fs: afero.NewReadOnlyFs(fs)}
}

func (r RosterService) LoadRosterRecords(path string) (map[string]string, error) {
        ZERO := make(map[string]string)

        fileBytes, err := afero.ReadFile(r.fs, path)
        if err != nil {
                return ZERO, err
        }

        var records []RosterRecord
        if err := csvutil.Unmarshal(fileBytes, &records); err != nil {
                return ZERO, err
        }

        // 姓の列の一意性チェック
        if len(lo.FindDuplicatesBy(records, func(r RosterRecord) string { return r.FamilyName })) > 0 {
                return ZERO, fmt.Errorf("姓の列で重複する値が存在します。")
        }

        // マップ化
        retMap := make(map[string]string)
        for _, record := range records {
                retMap[record.FamilyName] = record.FullName
        }

        return retMap, nil
}
